# 0. (OPCIONÁLIS) SDH-forrás előtisztítása — CSAK ha a felirat hang-/zene-
#    jegyzeteket tartalmaz ([music playing], [sighs], NÉV: címkék stb.)
#    Részletek: srt-preclean-addon\README.md
# py srt-preclean-addon\preclean_srt.py "input\Sorozat - S01E01.eng.srt"
# FONTOS: preclean után a 4. lépés ellenőrzését a .clean.srt ellen futtasd
# (az addon töröl cue-kat és ÚJRASZÁMOZ, ezért az eredeti .eng.srt-hez
# képest hamis hibákat jelezne), vagy használd az addon verify_preclean.py-át.

# 1. SRT szétvágás (ezt TE futtatod, nem kell hozzá Claude)
py split_srt.py "input\Sorozat - S01E01.eng.srt"
# Újra-splitnél (pl. más --block-size) a régi blokkok törléséhez: --clean
# (enélkül leáll, hogy a régi + új blokkok ne keveredjenek a merge-nél)

# 1.5 (OPCIONÁLIS, AJÁNLOTT) Glossary feltöltése MÉG A FORDÍTÁS ELŐTT
#     Csak az angol forrásból kinyeri a neveket / megszólításokat / helyszíneket,
#     az agent JAVASLATOT tesz a magyar fordításra (a CLAUDE.md szabályai alapján),
#     te jóváhagyod (y/n/e/q). Így a párhuzamos fordítás már EGYSÉGES nevekkel indul.
#     (Csak az angol fájlt add meg — a magyar argumentum elhagyása kapcsolja be ezt a módot.)
py glossary_extract.py "input\Sorozat - S01E01.eng.srt"

# 2a. Fordítás Claude Code-dal (ez indítja a párhuzamos agent-eket)
py translate_parallel.py "blocks\Sorozat - S01E01.eng" --agents 3

# 2b. ALTERNATÍVA: fordítás Gemini API-val (olcsóbb, ugyanazt a blocks/ mappát fordítja)
#     A két fordító felváltva is használható, mert ugyanúgy checkpoint-olnak.
#     Default modell: gemini-3.1-flash-lite
# py translate_with_gemini.py "blocks\Sorozat - S01E01.eng" --agents 3
# Opcionálisan:
#   --model gemini-3.1-flash      más Gemini modell
#   --block 3                     csak a 3. blokk újrafordítása (auto zero-pad)
#   --max-retries 6               több retry rate-limit/5xx esetén
# Előfeltétel: pip install google-genai python-dotenv pydantic
# Előfeltétel: .env fájl GEMINI_API_KEY=... bejegyzéssel

# 3. Összefűzés (megint csak sima Python)
py merge_srt.py "blocks\Sorozat - S01E01.eng" "output\Sorozat - S01E01.hun.srt"
# Hiányzó blokkoknál leáll és listázza őket; --force kapcsolóval mégis összefűz

# 4. Ellenőrzés — ✓ vagy ✗ jelzi ha valami nem stimmel
py verify_srt.py "input\Sorozat - S01E01.eng.srt" "output\Sorozat - S01E01.hun.srt"

# 5. Stilisztikai review Claude-dal (tükörfordítások, ragozási hibák)
py review_with_claude.py "output\Sorozat - S01E01.hun.srt"
# Default modell: sonnet
# Ha megtalálja az angol forrást (input\Sorozat - S01E01.eng.srt), automatikusan
# odaadja a modellnek szekciónként [EN] sorként — így kevesebb a téves találat.
# Opcionálisan:
#   --chunk-size 150            nagyobb chunkok = kevesebb Claude hívás
#   --model haiku|sonnet|opus   modell-választás (default sonnet)
#   --english "input\....eng.srt"  angol forrás kézi megadása
#   --no-english                angol forrás kihagyása
#   --start-chunk N             chunk-tartomány pl. megszakítás utáni pótlásra
#   --end-chunk N --suffix _part2  hogy ne írja felül a meglévő riportot
# Kimenet: output\Sorozat - S01E01.hun_REVIEW_CLAUDE.txt (olvasható riport)
#          output\Sorozat - S01E01.hun_REVIEW_CLAUDE.json (az apply_review.py-hoz)
# A javításokat az 5c. lépés (apply_review.py) viszi át interaktívan

# 5b. Stilisztikai review Gemini-vel (alternatív / kiegészítő, Te döntöd mikor futtatod)
py review_with_gemini.py "output\Sorozat - S01E01.hun.srt"
# Default modell: gemini-3.1-flash-lite
# Ha megtalálja az angol forrást (input\Sorozat - S01E01.eng.srt), automatikusan
# odaadja Gemini-nek szekciónként [EN] sorként — így kevesebb a téves találat.
# Opcionálisan:
#   --chunk-size 150            nagyobb chunkok = kevesebb API hívás
#   --model <név>               más Gemini modell (pl. gemini-3.1-flash, gemini-2.5-flash, gemini-3.1-pro-preview)
#   --english "input\....eng.srt"  angol forrás kézi megadása
#   --no-english                angol forrás kihagyása
#   --start-chunk N             chunk-tartomány pl. kvótahiba utáni pótlásra
#   --end-chunk N --suffix _part2  hogy ne írja felül a meglévő riportot
# Kimenet: output\Sorozat - S01E01.hun_REVIEW_GEMINI.txt (olvasható riport)
#          output\Sorozat - S01E01.hun_REVIEW_GEMINI.json (az apply_review.py-hoz)
# Előfeltétel: pip install google-genai python-dotenv pydantic
# Előfeltétel: .env fájl GEMINI_API_KEY=... bejegyzéssel

# 5c. Review-javaslatok alkalmazása (a két riportot összefésüli és deduplikálja)
py apply_review.py "output\Sorozat - S01E01.hun.srt"
# Automatikusan megtalálja a fájl melletti _REVIEW_CLAUDE / _REVIEW_GEMINI
# riportokat (a .json a kanonikus, a .txt-t is érti). Találatonként kérdez:
#   y = alkalmaz, 1..9 = adott variáns, e = kézi szerkesztés, n = kihagy,
#   q = kilépés (az addig alkalmazottak mentődnek)
# Az első íráskor .bak mentés készül az eredetiről.
# Csak listázás módosítás nélkül: --dry-run

# 6. Szójegyzék bővítése a KÉSZ fordításból (fordítás UTÁNI mód)
#     Itt MINDKÉT fájlt megadod → a "hu" a magyar feliratban TÉNYLEGESEN használt fordítás.
#     (Az 1.5 lépés a fordítás ELŐTT javasol; ez a lépés UTÓLAG, a kész párból tanul.)
py glossary_extract.py "input\Sorozat - S01E01.eng.srt" "output\Sorozat - S01E01.hun.srt"
# Interaktívan jóváhagyod a javasolt kifejezéseket (y/n/e/q)
# A glossary.json-t a translate_parallel.py ÉS mindkét review script automatikusan
# betölti és átadja a modellnek (system prompt-ként) a következő futáskor.

# 7. Szegmentálás — sorhossz + CPS (véglegesítő formázás, a kézi review-javítások UTÁN)
#     Csak Python stdlib, nincs telepítendő függőség. Kiegészíti a verify_srt.py-t
#     (az CPS-t jelzi, de sorhosszt/tördelést nem). Részletek: resegment_srt.md
# 7a. QA-riport (csak olvasás) — bármikor futtatható, ez a QA-kapu:
py resegment_srt.py report "output\Sorozat - S01E01.hun.srt"
# 7b. Sorhossz-tisztítás (időzítést NEM változtat) — a VÉGLEGES szövegen futtasd:
py resegment_srt.py reflow "output\Sorozat - S01E01.hun.srt" -o "output\Sorozat - S01E01.hun.reflow.srt"
# Opcionálisan:
#   --in-place                     a bemenet felülírása (új fájl helyett)
#   --split                        2 sorba nem férő cue-k idő-arányos bontása (cue-számot változtat)
#   --max-chars 42 --target-cps 17 plafonok (más csatorna = más érték)
#   --lang hu                      nyelvi töréslogika (kötőszavak, névelők)
